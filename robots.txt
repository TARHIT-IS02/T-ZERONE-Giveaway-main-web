# The public site is fully indexable. The Admin Panel and Lucky Draw consoles are
# separate deployments on their own hostnames, so they are governed by their own
# robots files — this one cannot and does not speak for them. Their real protection
# is Firebase Auth plus the admin allowlist in the Firestore rules, not this file.

User-agent: *
Allow: /

# The standalone game is a client-side sub-app with no indexable text content of its
# own; keeping it out of the index avoids thin-content pages competing with the
# giveaway landing page in search results. It stays fully reachable for real users.
Disallow: /game/

# index.html is served at "/" already — indexing it a second time under its literal
# filename would create a duplicate URL for the exact same document.
Disallow: /index.html

# Service worker script and PWA manifest are functional assets, not content — no
# reason for them to occupy crawl budget or ever appear in search results.
Disallow: /sw.js
Disallow: /manifest.json

# Common non-content query strings (cache-busting, UTM/social/ad click params) can
# make a crawler treat "/?v=2" or "/?utm_source=..." as a distinct URL from "/".
# Blocking the pattern keeps every campaign link consolidating onto the canonical "/".
Disallow: /*?*utm_
Disallow: /*?*fbclid
Disallow: /*?*gclid

# Known aggressive/low-value scraper bots get blocked outright rather than merely
# discouraged — they ignore crawl-delay and add load without any indexing benefit.
User-agent: AhrefsBot
Disallow: /

User-agent: SemrushBot
Disallow: /

User-agent: MJ12bot
Disallow: /

User-agent: DotBot
Disallow: /

Sitemap: https://t-zerone-giveaway.vercel.app/sitemap.xml
